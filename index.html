VS Code File 1

backend/schema.sql (MySQL 8.0 DDL & Seed)

CREATE DATABASE IF NOT EXISTS atelier_vera_db;
USE atelier_vera_db;

CREATE TABLE categories (
    category_id INT AUTO_INCREMENT PRIMARY KEY,
    slug VARCHAR(50) NOT NULL UNIQUE,
    name VARCHAR(100) NOT NULL,
    description TEXT
) ENGINE=InnoDB;

CREATE TABLE products (
    product_id INT AUTO_INCREMENT PRIMARY KEY,
    sku VARCHAR(32) NOT NULL UNIQUE,
    category_id INT NOT NULL,
    title VARCHAR(180) NOT NULL,
    price_usd DECIMAL(10, 2) NOT NULL,
    stock_qty INT NOT NULL DEFAULT 20,
    image_url VARCHAR(500) NOT NULL,
    is_featured BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (category_id) REFERENCES categories(category_id)
) ENGINE=InnoDB;

CREATE TABLE orders (
    order_id INT AUTO_INCREMENT PRIMARY KEY,
    order_number VARCHAR(32) NOT NULL UNIQUE,
    customer_name VARCHAR(120) NOT NULL,
    customer_email VARCHAR(180) NOT NULL,
    shipping_address TEXT NOT NULL,
    city VARCHAR(100) NOT NULL,
    postal_code VARCHAR(20) NOT NULL,
    total_usd DECIMAL(10, 2) NOT NULL,
    order_status VARCHAR(30) DEFAULT 'CONFIRMED',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;
VS Code File 2

backend/server.js (Express + mysql2 REST API)

const express = require('express');
const mysql = require('mysql2/promise');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(express.json());
app.use(express.static('.'));

const pool = mysql.createPool({
  host: 'localhost',
  user: 'root',
  password: 'your_mysql_password',
  database: 'atelier_vera_db'
});

app.get('/api/products', async (req, res) => {
  const [rows] = await pool.query(
    'SELECT p.*, c.name AS category FROM products p JOIN categories c ON p.category_id = c.category_id'
  );
  res.json(rows);
});

app.post('/api/orders', async (req, res) => {
  const { orderNumber, customerName, customerEmail, city, total } = req.body;
  const [result] = await pool.query(
    'INSERT INTO orders (order_number, customer_name, customer_email, city, total_usd) VALUES (?, ?, ?, ?, ?)',
    [orderNumber, customerName, customerEmail, city, total]
  );
  res.status(201).json({ orderId: result.insertId, orderNumber });
});

app.listen(3000, () => console.log('Server running on http://localhost:3000'));
